## Problem
A backup approval must not self-invalidate merely because the approval carrier or root pointer advances the repository HEAD.

HEAD_CHANGE_NE_AUTOMATIC_APPROVAL_SUPERSESSION=true
APPROVAL_TAIL_WRITE_NE_RUNTIME_STATE_CHANGE=true
POINTER_TAIL_WRITE_NE_RUNTIME_STATE_CHANGE=true

## Fingerprint binding
PLUGIN_BACKUP_APPROVAL_FRESHNESS_BINDING=RUNTIME_PLUGIN_STATE_FINGERPRINT
HEAD_REMAINS_PROVENANCE=true
SEMANTIC_AUTHORITY_HEAD_REMAINS_PROVENANCE=true

RUNTIME_PLUGIN_STATE_FINGERPRINT_INPUTS=
PLUGIN_ID |
PLUGIN_VERSION |
PLUGIN_RELEASE_ID |
REPO_TEXT_SOURCE_TREE_SHA |
BINARY_BINDING_MANIFEST_SHA_OR_NOT_APPLICABLE |
RUNTIME_SEMANTIC_BODY_HEAD |
PLUGIN_CURRENT_SEMANTIC_CLASS

FINGERPRINT_CHANGE_SUPERSEDES_APPROVAL=true
FINGERPRINT_UNCHANGED_GOVERNANCE_ONLY_HEAD_ADVANCE_MAY_PRESERVE_APPROVAL=true

## Supersession
APPROVAL_SUPERSEDED_IF=
PLUGIN_ID_CHANGED |
PLUGIN_VERSION_CHANGED |
PLUGIN_RELEASE_CHANGED |
TEXT_SOURCE_TREE_CHANGED |
BINARY_BINDING_CHANGED |
RUNTIME_SEMANTIC_BODY_CHANGED |
PLUGIN_CURRENT_SEMANTIC_CLASS_CHANGED |
FINGERPRINT_CANNOT_BE_REVERIFIED

APPROVAL_NOT_SUPERSEDED_BY=
APPROVAL_CARRIER_WRITE_ITSELF |
ROOT_POINTER_WRITE_ITSELF |
PROVENANCE_ONLY_HEAD_ADVANCE_WITH_REVERIFIED_IDENTICAL_FINGERPRINT

## Safety
